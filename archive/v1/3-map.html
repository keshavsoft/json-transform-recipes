Quick API Guide

      
Install via npm:

      
npm install map-json-by-json

      
      
Define your desired output template and map raw data into it:

      
import { mapJson } from "https://cdn.jsdelivr.net/gh/keshavsoft/map-json-by-json@main/src/v1/map.js";

const template = {
  orderId: "ORDER_NUMBER",
  orderDate: "TRANSACTION_DATE",
  lineItems: {
    $from: "ORDER_ITEMS_COLLECTION",
    product: "ITEM_NAME",
    subtotal: "TOTAL_AMOUNT"
  }
};

const clean = mapJson(template, rawData);
