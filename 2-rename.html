Quick API Guide

      
Install via npm:

      
npm install rename-json-keys

      
      
Import from CDN or package in Node.js / browser:

      
import { renameJsonKeys } from "https://cdn.jsdelivr.net/gh/keshavsoft/rename-json-keys@main/src/v1/rename.js";

// Call with positional arguments or { inData, inKeys } options object
const clean = renameJsonKeys(sourceData, {
  "VOUCHER_ID": "id",
  "TRANSACTION_DATE": "date",
  "INVENTORY_ENTRIES_LIST": "items"
});
